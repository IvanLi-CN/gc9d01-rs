void Lcd_enter_sleep(void)
{
	SPI_WriteComm(0x28);
	DelayMs(120);
	SPI_WriteComm(0x10);
	DelayMs(50);
}
void Lcd_exit_sleep(void)
{
	SPI_WriteComm(0x11);
	DelayMs(120) ;
	SPI_WriteComm(0x29);
}
